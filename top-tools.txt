import hydra
from omegaconf import DictConfig
from llmesh.main import execute_mesh, setup_cli_environment

# Import downstream packages so their @register_primitive decorators fire
# This makes llmesh aware of all available tools
try:
    import ictools
except ImportError:
    pass

try:
    import cybertools
except ImportError:
    pass

@hydra.main(config_path="conf", config_name="config", version_base=None)
def main(cfg: DictConfig):
    # Hand off execution entirely to llmesh
    execute_mesh(cfg)

def cli():
    setup_cli_environment()
    main()

if __name__ == "__main__":
    cli()
